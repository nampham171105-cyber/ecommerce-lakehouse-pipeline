docker exec -it spark-master /opt/spark/bin/pyspark --master spark://spark-master:7077 --packages org.apache.hadoop:hadoop-aws:3.3.4,com.amazonaws:aws-java-sdk-bundle:1.12.262 --conf spark.jars.ivy=/tmp/.ivy --conf spark.hadoop.fs.s3a.endpoint=http://minio:9000 --conf spark.hadoop.fs.s3a.access.key=minioadmin --conf spark.hadoop.fs.s3a.secret.key=minioadmin --conf spark.hadoop.fs.s3a.path.style.access=true --conf spark.hadoop.fs.s3a.impl=org.apache.hadoop.fs.s3a.S3AFileSystem
/* jar connector của minIO
-packages : báo spark tự động tải về 2 thư viện từ maven(cần khớp với bản build spark đi kèm)
-conf ivy: làm thư mục cache để tải packages về
-conf endpoint : báo spark khi thấy s3a thì gửi request tới địa chỉ này
-conf access & secret key : cặp khóa xác thực
-conf path.style : kết nối đặc thù minio
-conf impl : báo spark dùng đúng class java này để xử lý
*/

/*
Lí do chưa cài dockerfile và conf
- dùng -packages để các file jar từ internet mỗi lần chạy để thử nghiệm trước khi 
customer dockerfile và conf(build sẵn 1 dockerfile đầy đủ image)
*/

docker exec -it spark-master /opt/spark/bin/pyspark --master spark://spark-master:7077 --conf spark.jars.ivy=/tmp/.ivy2 --packages io.delta:delta-spark_2.12:3.2.0,org.apache.hadoop:hadoop-aws:3.3.4,com.amazonaws:aws-java-sdk-bundle:1.12.262 --conf spark.sql.extensions=io.delta.sql.DeltaSparkSessionExtension --conf spark.sql.catalog.spark_catalog=org.apache.spark.sql.delta.catalog.DeltaCatalog --conf spark.hadoop.fs.s3a.endpoint=http://minio:9000 --conf spark.hadoop.fs.s3a.access.key=minioadmin --conf spark.hadoop.fs.s3a.secret.key=minioadmin --conf spark.hadoop.fs.s3a.path.style.access=true --conf spark.hadoop.fs.s3a.impl=org.apache.hadoop.fs.s3a.S3AFileSystem
/*
-packages : gói của delta lake
-conf extensions và -conf catalog : kích hoạt SQL xử lý theo delta lake
*/

1. Ghi delta
data = [(1, "An", "Engineering", 1200), (2, "Binh", "Sales", 900)]
df = spark.createDataFrame(data, ["id", "name", "department", "salary"])
df.write.format("delta").mode("overwrite").save("s3a://spark-bucket/delta/employees")

2. Thực hành update
from delta.tables import DeltaTable
delta_table = DeltaTable.forPath(spark, "s3a://spark-bucket/delta/employees")
delta_table.update(
    condition="name = 'An'",
    set={"salary": "1500"}
)
delta_table.toDF().show()

3. Time Travel

df_old = spark.read.format("delta").option("versionAsOf", 0).load("s3a://spark-bucket/delta/employees")
df_old.show()

docker exec -it spark-master /opt/spark/bin/pyspark --master spark://spark-master:7077 --conf spark.jars.ivy=/tmp/.ivy2 --packages io.delta:delta-spark_2.12:3.2.0,org.apache.hadoop:hadoop-aws:3.3.4,com.amazonaws:aws-java-sdk-bundle:1.12.262 --conf spark.sql.extensions=io.delta.sql.DeltaSparkSessionExtension --conf spark.sql.catalog.spark_catalog=org.apache.spark.sql.delta.catalog.DeltaCatalog --conf spark.hadoop.hive.metastore.uris=thrift://hive-metastore:9083 --conf spark.sql.catalogImplementation=hive --conf spark.hadoop.fs.s3a.endpoint=http://minio:9000 --conf spark.hadoop.fs.s3a.access.key=minioadmin --conf spark.hadoop.fs.s3a.secret.key=minioadmin --conf spark.hadoop.fs.s3a.path.style.access=true --conf spark.hadoop.fs.s3a.impl=org.apache.hadoop.fs.s3a.S3AFileSystem
/*
Thêm 2 dòng mới vào cách kết nối của delta lake
-conf hive : báo delta lake khi cần tra cứu bảng, hãy hỏi metastore ở địa chỉ này
-conf catalog = hive : dùng hive làm catalog chính thức
*/

- cần cài các file java từ trên maven về thư mục hive-lib rồi gắn vào volumns của hive(vì spark submit không thể tự cài)
- cần thêm file hive-conf/core-site.xml chứa thông số minio vì hive chưa biết
- Cần thêm healcheck do hive cần db-hive bật lên thành công trước

# 1. Xóa database bị lỗi cũ
spark.sql("DROP DATABASE IF EXISTS companydb CASCADE")

# 2. Tạo lại database với LOCATION trỏ vào MinIO
spark.sql("CREATE DATABASE companydb LOCATION 's3a://spark-bucket/warehouse/companydb.db'")

# 3. Chuyển sang dùng database mới
spark.sql("USE companydb")

# 4. Tạo lại bảng Delta
spark.sql("""
    CREATE TABLE employees (
        id INT, name STRING, department STRING, salary DOUBLE
    ) USING delta
    LOCATION 's3a://spark-bucket/warehouse/employees'
""")

spark.sql("INSERT INTO employees VALUES (1, 'An', 'Engineering', 1200)")
spark.sql("SELECT * FROM employees").show()

# 5. Thoát spark kiểm tra xem bảng còn không (vì nếu lưu catalog trên minio thì chỉ có spark session hiện tại mới đọc được)
